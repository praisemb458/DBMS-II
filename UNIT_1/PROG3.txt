DECLARE
  pname VARCHAR2(50) := &pname;
   qty NUMBER := &qty;
   price NUMBER := &price;
   dis_per NUMBER := &discount;
   amount NUMBER;
   discount NUMBER;
BEGIN
 pname := &pname;
 amount := qty * price;
 discount := amount * dis_per/100;

    DBMS_OUTPUT.PUT_LINE('Product Name   : ' || pname);
    DBMS_OUTPUT.PUT_LINE('Total Amount   : Rs. ' || amount);
    DBMS_OUTPUT.PUT_LINE('Discount : Rs. ' || discount);
END;
/
