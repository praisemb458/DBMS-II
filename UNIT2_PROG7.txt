DECLARE

  CURSOR c_customer IS
    SELECT customer_id, customer_name, email FROM customer;
  
  v_id customer.customer_id%TYPE;
  v_name customer.customer_name%TYPE;
  v_email customer.email%TYPE;

BEGIN

  OPEN c_customer;

  LOOP
    FETCH c_customer INTO v_id, v_name, v_email;
    EXIT WHEN c_customer%NOTFOUND;
    DBMS_OUTPUT.PUT_LINE('ID: ' || v_id || ', Name: ' || v_name || ', Email: ' || v_email);

  END LOOP;

  CLOSE c_customer;

END;
/
