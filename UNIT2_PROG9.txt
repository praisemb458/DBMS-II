DECLARE

    CURSOR c_emp IS
        SELECT employee_id, first_name, last_name, salary
        FROM employees
        ORDER BY salary DESC;
        
    v_emp c_emp%ROWTYPE;

BEGIN
    OPEN c_emp;

    LOOP
        FETCH c_emp INTO v_emp;
        EXIT WHEN c_emp%NOTFOUND;
        
        DBMS_OUTPUT.PUT_LINE('ID: ' || v_emp.employee_id || 
                             ' | Name: ' || v_emp.first_name || ' ' || v_emp.last_name || 
                             ' | Salary: ' || v_emp.salary);

    END LOOP;
    CLOSE c_emp;
END;
/
