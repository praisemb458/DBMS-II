DECLARE
    -- Declare variables to store user input and calculations
    v_prod_name VARCHAR2(50) := '&Enter Product_Name';
    v_qty       NUMBER       := &Enter Quantity;
    v_price     NUMBER       := &Enter Price;
    v_total     NUMBER;
    v_discount  NUMBER;
    v_payable   NUMBER;
BEGIN
 Calculate total amount
  v_total := v_qty * v_price;
    
 Calculate a simple 10% discount
  v_discount := v_total * 0.10;
    
 Calculate final payable amount
  v_payable := v_total - v_discount;
    
    -- Display the results
    DBMS_OUTPUT.PUT_LINE('--- Invoice Details ---');
    DBMS_OUTPUT.PUT_LINE('Product Name   : ' || v_prod_name);
    DBMS_OUTPUT.PUT_LINE('Total Amount   : Rs. ' || v_total);
    DBMS_OUTPUT.PUT_LINE('Discount (10%) : Rs. ' || v_discount);
    DBMS_OUTPUT.PUT_LINE('Net Payable    : Rs. ' || v_payable);
END;
/
