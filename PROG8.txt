Write a PL/SQL block that displays the gross salary of employees as per input EID. (Consider an EMP table with EID, EName, Deptno, Deptname, Gender, Age, BasicSal) with appropriate data types.) Gross_Salary: BASICSAL + (DA + HRA + Medical)-PF

RULES: HRA=15% of basic, DA=50% of basic,
Medical= Rs.500, PF=10% of basic.

SET SERVEROUTPUT ON;

CREATE TABLE EMP (
    EID      NUMBER(5) PRIMARY KEY,
    EName    VARCHAR2(50),
    Deptno   NUMBER(3),
    Deptname VARCHAR2(30),
    Gender   CHAR(1),
    Age      NUMBER(3),
    BasicSal NUMBER(10, 2)
);


INSERT INTO EMP VALUES (101, 'James Bond', 10, 'IT', 'M', 30, 50000);
INSERT INTO EMP VALUES (102, 'Jason Bourne', 20, 'HR', 'F', 28, 60000);
COMMIT;


DECLARE
    
    v_eid  EMP.EID%TYPE := &Enter_EID; 
    
    v_ename  EMP.EName%TYPE;
    v_basicsal  EMP.BasicSal%TYPE;
    
    
    v_hra    NUMBER(10, 2);
    v_da     NUMBER(10, 2);
    v_medical  NUMBER(10, 2) := 500; 
    v_pf    NUMBER(10, 2);
    v_gross_sal NUMBER(10, 2);

BEGIN
    
    SELECT EName, BasicSal
    INTO v_ename, v_basicsal
    FROM EMP
    WHERE EID = v_eid;

    
    v_hra       := 0.15 * v_basicsal;  
    v_da        := 0.50 * v_basicsal;  
    v_pf        := 0.10 * v_basicsal;  
    
    -- Formula: Gross_Salary = BASICSAL + (DA + HRA + Medical) - PF
    v_gross_sal := v_basicsal + (v_da + v_hra + v_medical) - v_pf;

    DBMS_OUTPUT.PUT_LINE('Employee ID   : ' || v_eid);
    DBMS_OUTPUT.PUT_LINE('Employee Name : ' || v_ename);
    DBMS_OUTPUT.PUT_LINE('Basic Salary  : Rs. ' || TO_CHAR(v_basicsal, '699,456.00'));
    DBMS_OUTPUT.PUT_LINE('HRA (15%)     : Rs. ' || TO_CHAR(v_hra, '123,999.00'));
    DBMS_OUTPUT.PUT_LINE('DA (50%)      : Rs. ' || TO_CHAR(v_da, '458,599.00'));
    DBMS_OUTPUT.PUT_LINE('Medical       : Rs. ' || TO_CHAR(v_medical, '320,499.00'));
    DBMS_OUTPUT.PUT_LINE('PF (10%)      : Rs. ' || TO_CHAR(v_pf, '934,987.00'));
    DBMS_OUTPUT.PUT_LINE('GROSS SALARY  : Rs. ' || TO_CHAR(v_gross_sal, '129,569.00'));
  
END;
