DECLARE

  v_pattern VARCHAR2(20) := 'A%';

BEGIN

  FOR r_rec IN (SELECT employee_id, first_name, last_name 
                FROM employees 
                WHERE first_name LIKE v_pattern) 

  LOOP
    DBMS_OUTPUT.PUT_LINE('ID: ' || r_rec.employee_id || ' | Name: ' || r_rec.first_name || ' ' || r_rec.last_name);

  END LOOP;
END;
