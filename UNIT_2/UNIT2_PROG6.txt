DECLARE

    NO_DEPT_FOUND EXCEPTION;
    
    v_dept_no EMP.DEPT_NO%TYPE := &input_dept_no;
    
    v_records_found BOOLEAN := FALSE;
   
    CURSOR c_emp IS
        SELECT * 
        FROM emp 
        WHERE dept_no = v_dept_no;

BEGIN
  
    FOR r_emp IN c_emp LOOP
        v_records_found := TRUE;
        
        INSERT INTO emp_backup VALUES r_emp;
    END LOOP;
    
    IF NOT v_records_found THEN
        RAISE NO_DEPT_FOUND;
    END IF;
    
    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Successfully backed up employee records for Department: ' || v_dept_no);

EXCEPTION

    WHEN NO_DEPT_FOUND THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('Error: No records found for Department Number ' || v_dept_no);
        
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('An unexpected error occurred: ' || SQLERRM);
END;
/
