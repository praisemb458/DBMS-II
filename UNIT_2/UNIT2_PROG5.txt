DECLARE
  CURSOR c_dept IS
    SELECT department_id, department_name 
    FROM departments;
    
  CURSOR c_emp (p_dept_id NUMBER) IS
    SELECT first_name, salary AS basic_salary, (salary * 1.25) AS gross_salary
    FROM employees
    WHERE department_id = p_dept_id;
    
  v_dept_total_gross NUMBER;
BEGIN
  FOR r_dept IN c_dept LOOP
    DBMS_OUTPUT.PUT_LINE('Department: ' || r_dept.department_name);
    v_dept_total_gross := 0;
    
    FOR r_emp IN c_emp(r_dept.department_id) LOOP
      DBMS_OUTPUT.PUT_LINE('  Employee: ' || r_emp.first_name || ' | Basic Salary: ' || r_emp.basic_salary);
      v_dept_total_gross := v_dept_total_gross + r_emp.gross_salary;
    END LOOP;
    
    DBMS_OUTPUT.PUT_LINE('  Total Department Gross Salary: ' || v_dept_total_gross);
  
  END LOOP;
END;
