// Write a PL/SQL block which converts temperature from Celsius to Fahrenheit.


SET SERVEROUTPUT ON

DECLARE

    celsius      NUMBER := 25; 
    fahrenheit   NUMBER;

BEGIN
   
    fahrenheit := (celsius * 9 / 5) + 32;


    DBMS_OUTPUT.PUT_LINE('Celsius: ' || celsius || '°C');
    DBMS_OUTPUT.PUT_LINE('Fahrenheit: ' || fahrenheit || '°F');

END;
/
