Write a PL/SQL block to accept employee name from a user if it exists, display has/her basic salary otherwise display appropriate message using exception handling.

SET SERVER OUTPUTON;

DECLARE

v_emp_name	emp.name%TYPE := '&Enter_Employee_Name';
v_salary		emp.sal%TYPE;

BEGIN

SELECT sal
INTO v_salary
FROM emp
WHERE UPPER(ename) = UPPER(v_emp_name);

DBMS_OUTPUT.PUT_LINE('Employee Name : '  // v_emp_name);
DBMS_OUTPUT.PUT_LINE('Basic Salary : ' // v_salary);

EXCEPTION

WHEN NO_DATA_FOUND THEN
	DBMS_OUTPUT.PUT_LINE('Error : Employee " ' // v_emp_name // ' " does not exist ');

WHEN TOO_MANY_ROWS THEN
	DBMS_OUTPUT.PUT_LINE('Error : Multiple employees found with the name " ' // v_emp_name // ' " ');

WHEN OTHERS THEN
	DBMS_OUTPUT.PUT_LINE('Unexpected error occured: ' // SQLERRM); 

END;
/
