SET SERVEROUTPUT ON

DECLARE

cursor C1 IS

SELECT
FROM EMP
WHERE GENDER='Male'
AND DEPTNAME='HR';

R EMP%ROWTYPE;

BEGIN

OPEN C1;

LOOP

FETCH C1 INTO R;

EXIT WHEN C1%NOTFOUND;

DBMS_OUTPUT.PUT_LINE(

R.EID||''||
R.ENAME||''||
R.DEPTNAME||''||
R.BASICSAL);

END LOOP;

CLOSE C1;

END;
/
