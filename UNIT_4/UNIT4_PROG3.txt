CREATE OR REPLACE PROCEDURE SEARCH_EMP_BY_ID (
    P_EMPNO IN NUMBER,
    P_ENAME OUT VARCHAR2
) IS
BEGIN
    SELECT ENAME INTO P_ENAME
    FROM EMP
    WHERE EMPNO = P_EMPNO;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20001, 'Error: Employee ID not found.');
END;
/
