CREATE OR REPLACE FUNCTION FETCH_ACCOUNT_BALANCE (
    P_ACNO IN ACCOUNT.ACNO%TYPE
) RETURN ACCOUNT.BALANCE%TYPE IS
    V_BAL ACCOUNT.BALANCE%TYPE;
BEGIN
    SELECT BALANCE INTO V_BAL
    FROM ACCOUNT
    WHERE ACNO = P_ACNO;
    
    RETURN V_BAL;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20003, 'Invalid Account Number.');
END;
/
