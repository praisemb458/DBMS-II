CREATE OR REPLACE PROCEDURE SEARCH_EMP (
    P_EMPNO IN NUMBER,
    P_ENAME OUT VARCHAR2
) IS
BEGIN
    SELECT ENAME INTO P_ENAME
    FROM EMP
    WHERE EMPNO = P_EMPNO;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        P_ENAME := NULL;
        RAISE_APPLICATION_ERROR(-20002, 'Employee Number does not exist.');
END;
/
